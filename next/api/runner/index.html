    Runs programs on a share of this machine: one input, a list, or a session.

    Parameters:

    Name Type Description Default
    threads int | None

    The runner's thread count: the most threads its calculations run at once, and what a calculation that sets no threads gets. None: OMP_NUM_THREADS if set (a dispatcher's choice), else every core this process may use (its affinity and cgroup quota).

    		 None
    gpus Literal['auto'] | Sequence[str]

    GPUs it may use: "auto" for every visible GPU (CUDA_VISIBLE_DEVICES or nvidia-smi, see visible_gpus), or a list of device identifiers as for CUDA_VISIBLE_DEVICES; () for none. With "auto" and CUDA_VISIBLE_DEVICES unset, the runner sets CUDA_DEVICE_ORDER=PCI_BUS_ID (unless set), so CUDA numbers the GPUs as nvidia-smi does. A calculation gets whole GPUs.

    		 'auto'
    isolation Literal['auto', 'process']

    "auto" runs programs in this process unless they cannot share it (in_process=False), which run in worker processes; "process" runs every program in a worker process, so a crash ends only the worker.

    		 'auto'

    A runner never changes a number: it hands each calculation its thread count, GPUs and memory (qccompute.program.RunContext) and the program decides. It never runs more threads at once than its own count; work beyond it waits its turn (the docs' Threads page). Close a runner (or use it as a context manager) to stop its worker processes.

    def close(self) -> None:
    """Stop this runner's worker processes; it cannot run in workers afterwards.

    Submitted calculations that have not started are cancelled, and so is work
    waiting for the runner's threads. A worker still running a calculation (from
    another thread) is killed with every process it started; that calculation
    fails with kind ``environment``.
    """
    with self._submitting:
        futures, self._futures = self._futures, None
        waiting = [f for f in self._submitted if not f._control.started]
    for future in waiting:
        future.cancel()
    self._count.close()
    if futures is not None:
        futures.shutdown(wait=False, cancel_futures=True)
    with self._pools_changed:
        self._closed = True
        self._pools.clear()
        self._pools_changed.notify_all()
    for worker in list(self._all_workers):
        worker.close()

    def identities(self, inp: Input) -> dict[str, str]:
    """The numerics identity of the input's program and of every subprogram.

    ``{name: numerics}``, by name, recursively through the subprograms' own
    subprograms, from the programs this runner would run now, each asked with its
    own spec (`qccompute.program.numerics_of`). A program named with several specs
    gets its distinct identities sorted and joined by ``";"``
    (``"gfn2:5;gxtb-2.0.1:7"``), so the map stays ``{program: string}``. A program
    that is not installed is left out.

    Args:
        inp: The input.

    Returns:
        The identities the runner sets on the input's output
        (``execution.identities``), and a store files its run under.
    """
    if not isinstance(inp, ProgramInput):
        entry = self._entry(inp)
        return {} if entry is None else {FILES: numerics_of(entry.program, inp)}
    found: dict[str, set[str]] = {}
    pending: list[ProgramSpec] = [inp]
    while pending:
        spec = pending.pop()
        program = find_program(spec.program)
        if program is not None:
            found.setdefault(spec.program, set()).add(numerics_of(program, spec))
        pending.extend(spec.subprograms)
    return {name: ";".join(sorted(found[name])) for name in sorted(found)}

    def compute(
    self,
    inputs: Input | Sequence[Input],
    *,
    store: Store | None = None,
    timeout: float | None = None,
    rerun: bool = False,
    tags: Sequence[str] = (),
    meta: Mapping[str, Any] | None = None,
    raise_exc: bool = True,
    progress: ProgressCallback | None = None,
) -> Output | list[Output]:
    """Run one input, or a list of independent inputs side by side.

    Every input is planned first: an input its program refuses fails at once,
    without running. One calculation gets the runner's whole thread count unless
    it sets ``threads``; a list shares it, each input getting the count divided by
    the inputs that run (store hits don't count), and inputs of a program with
    batches go to it together in one call. Work beyond the runner's count waits
    its turn. The docs' Threads page explains the rules.

    Args:
        inputs: One input, or a list of them (of any programs).
        store: Look each input up first and return a reusable stored output; store
            every output that ran, with the identities of the programs that made
            it (`identities`). A stored run is reused only when its identities
            are those of the programs the runner would run now: a success always,
            a failure only when the input determines it (kind ``input``,
            ``unsupported`` or ``convergence``, or ``program`` with such a cause);
            ``rerun=True`` runs past it.
        timeout: Seconds each calculation may run, from when it starts; past it,
            the calculation fails with kind ``timeout``. Programs that start
            processes enforce it, and the runner kills programs in worker
            processes. In this process it is the program's to honour
            (``ctx.timeout``): the runner cannot stop a thread, and only starts
            none of its subprograms after it. For a hard limit on any program, use
            ``isolation="process"``.
        rerun: Run even when the store has a reusable output.
        tags: Tags for every key computed or reused (needs ``store``).
        meta: Metadata stored with each run (needs ``store``).
        raise_exc: Raise for failures (after every input of a list ran); False
            returns failed outputs in place.
        progress: Called with the events programs emit while they run (qcopt:
            one per step): small JSON objects, each with ``"path"``, the program
            names from the input down to the program that emitted it (a
            subprogram's events come through its parent). It may be called from
            several threads at once, and must be quick; an exception it raises
            is a warning. Stored outputs emit nothing; nothing is stored. A
            list's events do not say which input they are from: `submit` each
            input with a callable of its own for that.

    Returns:
        The output, or the outputs in input order.

    Raises:
        ProgramError: The calculation failed; its ``output`` is the failed
            output, with partial results and logs.
        ProgramErrors: Some calculations of a list failed; ``results`` holds
            every output in input order.
        RuntimeError: Called from a thread that is running a calculation or a
            session request (a program uses ``ctx.compute``).
    """
    check_free()
    if (tags or meta) and store is None:
        raise ValueError("tags and meta are stored with runs: pass a store.")
    if timeout is not None and not timeout > 0:
        raise ValueError("timeout must be a positive number of seconds.")
    if isinstance(inputs, ProgramInput | FileInput):
        out, failure = self._one(
            inputs, store, timeout, rerun, tags, meta, progress
        )
        if failure is not None and raise_exc:
            raise failure
        return out
    inputs = list(inputs)
    outputs: list[Output | None] = [None] * len(inputs)
    failures: dict[int, ProgramError] = {}
    listed = self._run_list(
        inputs, store, rerun, tags, meta, self._top(), timeout, None, progress
    )
    for i, out, failure in listed:
        outputs[i] = out
        if failure is not None:
            failures[i] = failure
    done = [o for o in outputs if o is not None]
    if failures and raise_exc:
        raise ProgramErrors(
            f"{len(failures)} of {len(inputs)} calculations failed",
            [failures[i] for i in sorted(failures)],
            done,
        )

    qccompute.serve ¶

    The worker: one loop that hosts any program's sessions in its own process.

    python -m qccompute.serve mychem                  # stdin/stdout, one client
python -m qccompute.serve mychem --socket PATH    # a Unix socket, one client at a time
python -m qccompute.serve mychem --scratch DIR    # temporary scratch made in DIR

    A qccompute.worker.Worker gives its worker a --scratch directory of its own and removes it when the worker ends, so a killed worker leaves no scratch behind.

    A worker keeps a program alive between calls (warm starts, a live GPU context) and isolates it: a crash ends the worker, not its caller. The program is found by name (qccompute.program.find_program: registered in this process, else the qccompute.programs entry points), given as module:attribute, or as pickle:PATH, a file holding it pickled (a runner's, for a program registered in its process that no module holds). The loop uses only qcdata and the protocol, and knows nothing of the program beyond it. Design: docs/design/program-interface.md, "Workers".

    Wire, one JSON message per line:

    • The first line is the hello: {"protocol": 1, "program", "version", "capabilities", "array_files": true, "calls": true}. array_files and calls say what the options message may turn on.
    • Every message carries an id; replies are {"id", "ok": true, "result"} or {"id", "ok": false, "error": ErrorInfo}. ok: false is for the message itself (bad JSON, an unknown op or session, a failed open or plan); a failed calculation is a result: a failed Frame or ProgramOutput.
    • Ops: open {input, calctypes, ctx} -> {"session", "program_version"}; compute {session, positions, calctype, accuracy, lineage, outputs, seed, commit} -> Frame; compute_many {session, requests} -> [Frame]; reset {session, lineage} (absent: every lineage; null: the default one); plan {session}, {session, calctype} or {input}; close {session}; compute {input, ctx} (a one-off) -> ProgramOutput; compute_batch {inputs, ctx} -> [ProgramOutput]; lineages {session} where the session offers them; options {array_dir, calls}.
    • ctx is {threads, memory_gib, device_memory_gib, timeout, scratch_dir, gpus, progress, stop_file, identities}. The GPU is fixed when the worker is spawned, so gpus must be empty or match CUDA_VISIBLE_DEVICES. stop_file names a file whose existence asks the calculation to stop at its next step (ctx.stop_requested()): the client creates it while the message is in flight, and removes it after. While the message runs, the worker holds its BLAS and OpenMP pools at threads (at one thread for an input that names subprograms, whose grant it is). With progress: true the program's progress events (ctx.progress), each with its path from the worker's program down ([] for its own; a subprogram's name for one the worker ran itself), stream to the client while a message is in flight, as lines {"id", "progress": event} naming the message (for a session, each request's message); an event that is not JSON is dropped, with a note on stderr. A plain client never asks for them and never gets one.
    • Numbers are shortest round-trip decimals, so positions and results cross bit for bit. No NaN or Infinity: a message holding one is refused, and a non-finite result becomes a failed frame (kind numerical, code non_finite).

    • options sets what this connection uses; a plain JSON-lines client never sends it and gets neither:

    • array_dir: a directory both sides can read and write (workers are local). An array field over 1 MiB, in a message either way, crosses as a .npy file there, with qcdata's pointer {path, shape, dtype, sha256} in its place (the format of a saved model's arrays); the reader checks it and deletes the file.

    • calls: true: the client runs the subprograms of this worker's program (its ctx.compute and ctx.session) that the worker cannot run exactly as the client would, so they see the programs the client sees, registered ones included, and share its grant. The worker runs a call itself when every program its input names is found here with the identities the client gave in the message's ctx, may run in process, and its plan uses no GPU. While a message is in flight the worker may write call lines, {"call", "op", ...} with the ops above (compute {input}, compute_batch {inputs}, open {input, calctypes, ctx: {threads}} and the session ops); the client answers each on stdin, {"call", "ok", "result"} or {"call", "ok": false, "error"}, in any order. Without it, subprograms run in the worker's own default runner.
    • End of input closes every session and exits 0. After a reply whose error has the code device_lost (a GPU fault that poisoned the device context) the worker exits 3: nothing more can run in this process.

    Hygiene: stdout carries protocol lines only. At start the worker writes the protocol to a dup of the original stdout and points file descriptor 1 at stderr, because CUDA, BLAS and other libraries print.

    def load_program(spec: str) -> Program:
    """The program a spec names.

    A name (`find_program`), the object at ``module:attribute``, or ``pickle:PATH``:
    a file holding the program pickled, which a runner writes for a program
    registered in its process that no module holds (`qccompute.runner._worker_spec`).

    Raises:
        SystemExit: No such program.
    """
    if spec.startswith("pickle:"):
        with open(spec.removeprefix("pickle:"), "rb") as f:
            loaded: Program = pickle.load(f)
        return loaded
    if ":" in spec:
        module, attr = spec.split(":", 1)
        program: Program = getattr(importlib.import_module(module), attr)
        return program
    found = find_program(spec)
    if found is None:
        raise SystemExit(
            f"qccompute.serve: no program {spec!r} (registered or in the "
            "qccompute.programs entry points)"
        )
    return found

    def serve(host: Host, lines: Iterable[str], write: Callable[[str], None]) -> int:
    """Answer every message of ``lines`` through ``write``, after the hello.

    Each call is one connection: its ``options`` start unset.

    Args:
        host: The program's sessions.
        lines: Message lines.
        write: Writes one reply line (without its newline).

    Returns:
        The exit status: 0 at the end of input, `EXIT_DEVICE_LOST` after a
        ``device_lost`` reply.
    """
    connection = _Connection(iter(lines), write)
    host.connection = connection
    host.array_dir = None
    connection.write(_wire.encode(host.hello()))
    for line in connection.messages():
        if not line.strip():
            continue
        reply = _reply(host, line)
        if reply is None:
            continue
        connection.write(reply)
        if host.device_lost:
            return EXIT_DEVICE_LOST
    return 0

    def main(argv: list[str] | None = None) -> int:
    """Run a worker; the exit status.

    Args:
        argv: Command-line arguments (default: ``sys.argv``).
    """
    parser = argparse.ArgumentParser(
        prog="python -m qccompute.serve",
        description="Host a program's sessions in this process, over JSON lines.",
    )
    parser.add_argument("program", help="a program name, or module:attribute")
    parser.add_argument(
        "--socket", help="serve a Unix socket at this path, one client at a time"
    )
    parser.add_argument(
        "--scratch",
        help="make calculations' temporary scratch directories in this directory, "
        "which the client removes when the worker ends or is killed",
    )
    args = parser.parse_args(argv)
    paths = os.environ.pop("QCCOMPUTE_WORKER_SYS_PATH", "")
    sys.path.extend(p for p in paths.split(os.pathsep) if p and p not in sys.path)
    scratch = Path(args.scratch) if args.scratch else None
    if args.socket is None:
        proto = _protocol_stdout()
        host = Host(load_program(args.program), scratch=scratch)
        try:
            return serve(host, _lines(sys.stdin), _writer(proto))
        finally:
            host.close_all()
    host = Host(load_program(args.program), scratch=scratch)
    sys.stdout = sys.stderr
    path = Path(args.socket)
    path.unlink(missing_ok=True)
    server = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
    server.bind(str(path))
    server.listen()
    try:
        while True:
            conn, _ = server.accept()
            with (
                conn,
                conn.makefile("r", encoding="utf-8") as reader,
                conn.makefile("w", encoding="utf-8") as writer,
            ):
                status = serve(host, _lines(reader), _writer(writer))
            if status:
                return status
    finally:
        host.close_all()
        server.close()
        path.unlink(missing_ok=True)
